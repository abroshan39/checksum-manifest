cmake_minimum_required(VERSION 3.10)
project(checksum_manifest VERSION 1.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# Dependency overrides. An exact *_LIBRARY file takes precedence over the
# corresponding *_LIBRARY_DIR. Leave all overrides empty for normal discovery.
set(CM_OPENSSL_INCLUDE_DIR "" CACHE PATH "OpenSSL 1.1.0+ include directory containing openssl/evp.h")
set(CM_OPENSSL_LIBRARY_DIR "" CACHE PATH "Directory containing the OpenSSL Crypto library")
set(CM_OPENSSL_LIBRARY "" CACHE FILEPATH "Exact OpenSSL Crypto library file")
set(CM_GCRYPT_INCLUDE_DIR "" CACHE PATH "Libgcrypt include directory containing gcrypt.h")
set(CM_GCRYPT_LIBRARY_DIR "" CACHE PATH "Directory containing the Libgcrypt library")
set(CM_GCRYPT_LIBRARY "" CACHE FILEPATH "Exact Libgcrypt library file")
set(CM_RHASH_INCLUDE_DIR "" CACHE PATH "LibRHash include directory containing rhash.h")
set(CM_RHASH_LIBRARY_DIR "" CACHE PATH "Directory containing the LibRHash library")
set(CM_RHASH_LIBRARY "" CACHE FILEPATH "Exact LibRHash library file")

# GCC/Clang users commonly use CPATH/C_INCLUDE_PATH and LIBRARY_PATH for direct
# compiler builds. CMake's find_* commands do not natively treat CPATH or
# LIBRARY_PATH as CMake search variables, so import them as additional hints.
set(CM_ENV_INCLUDE_HINTS "")
foreach(CM_ENV_INCLUDE_VAR CPATH C_INCLUDE_PATH)
    if(DEFINED ENV{${CM_ENV_INCLUDE_VAR}} AND NOT "$ENV{${CM_ENV_INCLUDE_VAR}}" STREQUAL "")
        file(TO_CMAKE_PATH "$ENV{${CM_ENV_INCLUDE_VAR}}" CM_ENV_INCLUDE_PATHS)
        list(APPEND CM_ENV_INCLUDE_HINTS ${CM_ENV_INCLUDE_PATHS})
    endif()
endforeach()
list(REMOVE_DUPLICATES CM_ENV_INCLUDE_HINTS)

set(CM_ENV_LIBRARY_HINTS "")
if(DEFINED ENV{LIBRARY_PATH} AND NOT "$ENV{LIBRARY_PATH}" STREQUAL "")
    file(TO_CMAKE_PATH "$ENV{LIBRARY_PATH}" CM_ENV_LIBRARY_PATHS)
    list(APPEND CM_ENV_LIBRARY_HINTS ${CM_ENV_LIBRARY_PATHS})
endif()
list(REMOVE_DUPLICATES CM_ENV_LIBRARY_HINTS)

# Also expose the imported compiler search paths to FindOpenSSL and any other
# CMake find module used by this project.
list(APPEND CMAKE_INCLUDE_PATH ${CM_ENV_INCLUDE_HINTS})
list(APPEND CMAKE_LIBRARY_PATH ${CM_ENV_LIBRARY_HINTS})

add_executable(checksum-manifest checksum-manifest.c)
set(CM_PRIVATE_INCLUDE_DIRS "")
set(CM_PRIVATE_LIBRARIES "")

# OpenSSL: use FindOpenSSL when no explicit project override is supplied. With
# any override, resolve header/library explicitly so custom SDK layouts remain
# predictable.
if(NOT CM_OPENSSL_INCLUDE_DIR AND NOT CM_OPENSSL_LIBRARY_DIR AND NOT CM_OPENSSL_LIBRARY)
    find_package(OpenSSL 1.1.0 REQUIRED)
    list(APPEND CM_PRIVATE_LIBRARIES OpenSSL::Crypto)
else()
    if(CM_OPENSSL_INCLUDE_DIR)
        if(NOT EXISTS "${CM_OPENSSL_INCLUDE_DIR}/openssl/evp.h")
            message(FATAL_ERROR "openssl/evp.h not found below CM_OPENSSL_INCLUDE_DIR='${CM_OPENSSL_INCLUDE_DIR}'")
        endif()
        list(APPEND CM_PRIVATE_INCLUDE_DIRS "${CM_OPENSSL_INCLUDE_DIR}")
        set(CM_OPENSSL_INCLUDE_DIR_RESOLVED "${CM_OPENSSL_INCLUDE_DIR}")
    else()
        find_path(CM_OPENSSL_INCLUDE_DIR_AUTO NAMES openssl/evp.h HINTS ${CM_ENV_INCLUDE_HINTS})
        if(NOT CM_OPENSSL_INCLUDE_DIR_AUTO)
            message(FATAL_ERROR "OpenSSL headers not found. Set CM_OPENSSL_INCLUDE_DIR or add the include directory to CPATH/C_INCLUDE_PATH/CMAKE_INCLUDE_PATH.")
        endif()
        list(APPEND CM_PRIVATE_INCLUDE_DIRS "${CM_OPENSSL_INCLUDE_DIR_AUTO}")
        set(CM_OPENSSL_INCLUDE_DIR_RESOLVED "${CM_OPENSSL_INCLUDE_DIR_AUTO}")
    endif()

    include(CheckCSourceCompiles)
    unset(CM_OPENSSL_VERSION_SUPPORTED CACHE)
    set(CM_SAVED_REQUIRED_INCLUDES "${CMAKE_REQUIRED_INCLUDES}")
    set(CMAKE_REQUIRED_INCLUDES "${CM_OPENSSL_INCLUDE_DIR_RESOLVED}")
    check_c_source_compiles("#include <openssl/opensslv.h>
#if OPENSSL_VERSION_NUMBER < 0x10100000L
#error OpenSSL 1.1.0 or newer is required
#endif
int main(void) { return 0; }" CM_OPENSSL_VERSION_SUPPORTED)
    set(CMAKE_REQUIRED_INCLUDES "${CM_SAVED_REQUIRED_INCLUDES}")
    if(NOT CM_OPENSSL_VERSION_SUPPORTED)
        message(FATAL_ERROR "OpenSSL 1.1.0 or newer is required.")
    endif()

    if(CM_OPENSSL_LIBRARY)
        if(NOT EXISTS "${CM_OPENSSL_LIBRARY}")
            message(FATAL_ERROR "CM_OPENSSL_LIBRARY does not exist: '${CM_OPENSSL_LIBRARY}'")
        endif()
        list(APPEND CM_PRIVATE_LIBRARIES "${CM_OPENSSL_LIBRARY}")
    else()
        if(CM_OPENSSL_LIBRARY_DIR)
            find_library(CM_OPENSSL_LIBRARY_AUTO NAMES crypto libcrypto PATHS "${CM_OPENSSL_LIBRARY_DIR}" NO_DEFAULT_PATH)
        else()
            find_library(CM_OPENSSL_LIBRARY_AUTO NAMES crypto libcrypto HINTS ${CM_ENV_LIBRARY_HINTS})
        endif()
        if(NOT CM_OPENSSL_LIBRARY_AUTO)
            message(FATAL_ERROR "OpenSSL Crypto library not found. Set CM_OPENSSL_LIBRARY/CM_OPENSSL_LIBRARY_DIR or add the library directory to LIBRARY_PATH/CMAKE_LIBRARY_PATH.")
        endif()
        list(APPEND CM_PRIVATE_LIBRARIES "${CM_OPENSSL_LIBRARY_AUTO}")
    endif()
endif()

if(CM_GCRYPT_INCLUDE_DIR)
    if(NOT EXISTS "${CM_GCRYPT_INCLUDE_DIR}/gcrypt.h")
        message(FATAL_ERROR "gcrypt.h not found in CM_GCRYPT_INCLUDE_DIR='${CM_GCRYPT_INCLUDE_DIR}'")
    endif()
    list(APPEND CM_PRIVATE_INCLUDE_DIRS "${CM_GCRYPT_INCLUDE_DIR}")
else()
    find_path(CM_GCRYPT_INCLUDE_DIR_AUTO NAMES gcrypt.h HINTS ${CM_ENV_INCLUDE_HINTS})
    if(NOT CM_GCRYPT_INCLUDE_DIR_AUTO)
        message(FATAL_ERROR "Libgcrypt header gcrypt.h not found. Set CM_GCRYPT_INCLUDE_DIR or add the include directory to CPATH/C_INCLUDE_PATH/CMAKE_INCLUDE_PATH.")
    endif()
    list(APPEND CM_PRIVATE_INCLUDE_DIRS "${CM_GCRYPT_INCLUDE_DIR_AUTO}")
endif()

if(CM_GCRYPT_LIBRARY)
    if(NOT EXISTS "${CM_GCRYPT_LIBRARY}")
        message(FATAL_ERROR "CM_GCRYPT_LIBRARY does not exist: '${CM_GCRYPT_LIBRARY}'")
    endif()
    list(APPEND CM_PRIVATE_LIBRARIES "${CM_GCRYPT_LIBRARY}")
else()
    if(CM_GCRYPT_LIBRARY_DIR)
        find_library(CM_GCRYPT_LIBRARY_AUTO NAMES gcrypt libgcrypt-20 PATHS "${CM_GCRYPT_LIBRARY_DIR}" NO_DEFAULT_PATH)
    else()
        find_library(CM_GCRYPT_LIBRARY_AUTO NAMES gcrypt libgcrypt-20 HINTS ${CM_ENV_LIBRARY_HINTS})
    endif()
    if(NOT CM_GCRYPT_LIBRARY_AUTO)
        message(FATAL_ERROR "Libgcrypt library not found. Set CM_GCRYPT_LIBRARY/CM_GCRYPT_LIBRARY_DIR or add the library directory to LIBRARY_PATH/CMAKE_LIBRARY_PATH.")
    endif()
    list(APPEND CM_PRIVATE_LIBRARIES "${CM_GCRYPT_LIBRARY_AUTO}")
endif()

if(CM_RHASH_INCLUDE_DIR)
    if(NOT EXISTS "${CM_RHASH_INCLUDE_DIR}/rhash.h")
        message(FATAL_ERROR "rhash.h not found in CM_RHASH_INCLUDE_DIR='${CM_RHASH_INCLUDE_DIR}'")
    endif()
    list(APPEND CM_PRIVATE_INCLUDE_DIRS "${CM_RHASH_INCLUDE_DIR}")
else()
    find_path(CM_RHASH_INCLUDE_DIR_AUTO NAMES rhash.h HINTS ${CM_ENV_INCLUDE_HINTS})
    if(NOT CM_RHASH_INCLUDE_DIR_AUTO)
        message(FATAL_ERROR "LibRHash header rhash.h not found. Set CM_RHASH_INCLUDE_DIR or add the include directory to CPATH/C_INCLUDE_PATH/CMAKE_INCLUDE_PATH.")
    endif()
    list(APPEND CM_PRIVATE_INCLUDE_DIRS "${CM_RHASH_INCLUDE_DIR_AUTO}")
endif()

if(CM_RHASH_LIBRARY)
    if(NOT EXISTS "${CM_RHASH_LIBRARY}")
        message(FATAL_ERROR "CM_RHASH_LIBRARY does not exist: '${CM_RHASH_LIBRARY}'")
    endif()
    list(APPEND CM_PRIVATE_LIBRARIES "${CM_RHASH_LIBRARY}")
else()
    if(CM_RHASH_LIBRARY_DIR)
        find_library(CM_RHASH_LIBRARY_AUTO NAMES rhash librhash PATHS "${CM_RHASH_LIBRARY_DIR}" NO_DEFAULT_PATH)
    else()
        find_library(CM_RHASH_LIBRARY_AUTO NAMES rhash librhash HINTS ${CM_ENV_LIBRARY_HINTS})
    endif()
    if(NOT CM_RHASH_LIBRARY_AUTO)
        message(FATAL_ERROR "LibRHash library not found. Set CM_RHASH_LIBRARY/CM_RHASH_LIBRARY_DIR or add the library directory to LIBRARY_PATH/CMAKE_LIBRARY_PATH.")
    endif()
    list(APPEND CM_PRIVATE_LIBRARIES "${CM_RHASH_LIBRARY_AUTO}")
endif()

target_include_directories(checksum-manifest PRIVATE ${CM_PRIVATE_INCLUDE_DIRS})
target_link_libraries(checksum-manifest PRIVATE ${CM_PRIVATE_LIBRARIES})

if(MSVC)
    target_compile_options(checksum-manifest PRIVATE /W4)
else()
    target_compile_options(checksum-manifest PRIVATE -Wall -Wextra -Wpedantic)
endif()

install(TARGETS checksum-manifest RUNTIME DESTINATION bin)
